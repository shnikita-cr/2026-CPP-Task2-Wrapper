cmake_minimum_required(VERSION 3.20)
project(WrapperEngineProject LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(text_processor src/text_processor.cpp)
target_include_directories(text_processor PUBLIC src)
if(MSVC)
    target_compile_options(text_processor PUBLIC /utf-8 /W4)
else()
    target_compile_options(text_processor PUBLIC -Wall -Wextra -Werror)
endif()
enable_testing()
add_executable(wrapper_app src/main.cpp)
target_link_libraries(wrapper_app PRIVATE text_processor)
add_test(NAME BaseExecutionTest COMMAND wrapper_app)
